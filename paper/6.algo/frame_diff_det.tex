\noindent
Let $F_t$ denote the grayscale frame at time $t$.
$\mathbf{C}_t \in \{0, 255\}$ is the binary motion mask, where a pixel is declared
active if the absolute inter-frame intensity change exceeds threshold $\tau_d$.

\begin{algorithm}
\caption{Frame Difference Detector (FrameDiffDet)}
\label{alg:framediff_det}
\begin{algorithmic}[1]

\Require Frames $F_t, F_{t-1}$
\Require $\tau_d$: diff sensitivity
\Ensure  $\mathbf{C}_t \in \{0, 255\}$: binary motion mask

\Statex

\If{$F_{t-1} = \varnothing$}
    \State $F_{t-1} \leftarrow F_t$
    \State \Return $\mathbf{0}$
\EndIf

\Statex

\State $\mathbf{C}_t(i,j) \leftarrow
    \begin{cases}
        255 & |F_t(i,j) - F_{t-1}(i,j)| > \tau_d \\
        0   & \text{otherwise}
    \end{cases}$

\State $F_{t-1} \leftarrow F_t$

\State \Return $\mathbf{C}_t$

\end{algorithmic}
\end{algorithm}
