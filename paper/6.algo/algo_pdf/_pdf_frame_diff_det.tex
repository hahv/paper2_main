% ! Frame Difference Detector (FrameDiffDet)
\begin{algorithm}[H]
    \caption{Frame Difference Detector (FrameDiffDet)}
    \label{alg:framediff_det}
    \begin{algorithmic}[2]

        \Require Downsampled frames $\tilde{f}_t,\, \tilde{f}_{\text{prev}}$
        \Require $\tau_d$: pixel difference sensitivity threshold
        \Ensure $\mathbf{M}_t \in \{0, 255\}$: binary foreground mask

        \Statex

        \State Convert $\tilde{f}_t$ and $\tilde{f}_{\text{prev}}$ to grayscale:
               $F_t,\, F_{\text{prev}}$

        \Statex

        \State Compute binary mask, where each pixel $(i,j)$ is set as:
        \Statex \quad $\mathbf{M}_t(i,j) \leftarrow
            \begin{cases}
                255 & \text{if } |F_t(i,j) - F_{\text{prev}}(i,j)| > \tau_d \\
                0   & \text{otherwise}
            \end{cases}$

        \State \Return $\mathbf{M}_t$

    \end{algorithmic}
\end{algorithm}
