\documentclass[tikz,border=5pt]{standalone}
\usepackage{tikz}
\usepackage{booktabs}
\usepackage{amsmath}
\usetikzlibrary{automata, positioning, arrows.meta}
\begin{document}
\begin{tikzpicture}[
        node distance=8cm,
        every state/.style={draw, rounded corners, minimum width=2.6cm,
            minimum height=1.3cm, font=\small\bfseries},
        thick,
        >={Stealth[length=8pt, width=6pt]},
        auto
    ]

    %% ── States ───────────────────────────────────────────────────────
    \node[state, fill=blue!30, align=center] (normal) {NORMAL};
    \node[state, initial, initial where=left,
          initial text={\textbf{Start}},
          fill=orange!35, align=center,
          right of=normal] (eager) {EAGER};

    %% ── T1: EAGER → NORMAL ───────────────────────────────────────────
    \draw[->, line width=1.5pt] (eager) edge[bend left=25]
        node[below, align=center]{%
            \footnotesize
            \textit{if:}\; $c_{\mathrm{clear}} \geq W_{\mathrm{clr}}$\\[2pt]
            \textit{then:}\;
            $c_{\mathrm{fire}} \leftarrow 0,\;
             c_{\mathrm{skip}} \leftarrow 0$}
        (normal);

    %% ── T2: NORMAL → EAGER (Fixed: Now a black arrow) ────────────────
    \draw[->, line width=1.5pt, black] (normal) edge[bend left=25]
        node[above, align=center]{%
            \footnotesize
            \textit{if:}\; $c_{\mathrm{fire}} \geq W_{\mathrm{fire}}$ \\[2pt]
            \textit{then:}\; $c_{\mathrm{clear}} \leftarrow 0$}
        (eager);

    %% ── NORMAL top loop: skip ────────────────────────────────────────
    \draw[->] (normal) edge[loop above, looseness=5]
    node[above, align=center]{%
        \footnotesize
        \textit{if:}\;
        $s_t{=}0$ \textbf{and}
        $c_{\mathrm{skip}} < N_{\mathrm{chk}}$ \textbf{and}
        $c_{\mathrm{burst}}{=}0$ \\[2pt]
        \textit{then:}\; $c_{\mathrm{skip}} \mathrel{+}= 1$,\;
        $c_{\mathrm{fire}} \leftarrow 0$
        }
    (normal);

    %% ── NORMAL left loop: motion OR burst (M invoked) ────────────────
    \draw[->, red!70!black] (normal) edge[loop left, looseness=5]
        node[left, inner sep=8pt]{%
            \footnotesize
            \begin{tabular}{@{}r@{\;}l@{}}
                \textit{if:}   & $s_t=1$ \textbf{or} $c_{\mathrm{burst}} > 0$ \\[2pt]
                \textit{then:} & $c_{\mathrm{skip}} \leftarrow 0$ \\
                               & $c_{\mathrm{burst}} \leftarrow \max(0, c_{\mathrm{burst}} - 1)$ \\[2pt]
                               & $\hat{y}_t = 1 \implies c_{\mathrm{fire}} \mathrel{+}= 1$ \\
                               & $\hat{y}_t = 0 \implies c_{\mathrm{fire}} \leftarrow 0$
            \end{tabular}}
        (normal);

    %% ── NORMAL bottom loop: trigger burst (Fixed: Black arrow) ───────
    \draw[->, black] (normal) edge[loop below, looseness=5]
        node[below, align=center]{%
            \footnotesize
            \textbf{Trigger burst:}\\[2pt]
            \textit{if:}\; $c_{\mathrm{skip}} \geq N_{\mathrm{chk}}$ \\[2pt]
            \textit{then:}\; $c_{\mathrm{skip}} \leftarrow 0$,\;
                $c_{\mathrm{burst}} \leftarrow W_{\mathrm{fire}}$}
        (normal);

    %% ── EAGER top loop: positive ─────────────────────────────────────
    \draw[->, red!70!black] (eager) edge[loop above, looseness=5]
        node[above, align=center]{%
            \footnotesize
            \textit{if:}\; $\hat{y}_t{=}1$ \\[2pt]
            \textit{then:}\; $c_{\mathrm{clear}} \leftarrow 0$}
        (eager);

    %% ── EAGER bottom loop: negative (Fixed: Now a red arrow) ─────────
    \draw[->, red!70!black] (eager) edge[loop below, looseness=5]
        node[below, align=center]{%
            \footnotesize
            \textit{if:}\; $\hat{y}_t{=}0$ \textbf{and}
            $c_{\mathrm{clear}} < W_{\mathrm{clr}}$ \\[2pt]
            \textit{then:}\; $c_{\mathrm{clear}} \mathrel{+}= 1$}
        (eager);

    %% ── Legend ───────────────────────────────────────────────────────
    \node[draw, rounded corners, fill=gray!8,
          below=3.2cm of normal,
          anchor=north west, xshift=-5.5cm,
          text width=14cm, font=\footnotesize] (legend) {%
        \begin{tabular}{@{}cp{11.5cm}@{}}
            \multicolumn{2}{l}{\textbf{Operational Modes}} \\
            \midrule
            \tikz\draw[fill=blue!30, draw=black, rounded corners=2pt]
                (0,0) rectangle (0.9em, 0.9em);
            ~\textbf{NORMAL}
                & \textbf{Skip Module enabled.} Model $\mathcal{M}$ is invoked only if motion is detected ($s_t=1$) or a forced burst is active. \\[4pt]
            \tikz\draw[fill=orange!35, draw=black, rounded corners=2pt]
                (0,0) rectangle (0.9em, 0.9em);
            ~\textbf{EAGER}
                & \textbf{Continuous inference.} Skip module $\mathcal{S}$ is bypassed; $\mathcal{M}$ evaluates every frame. (Default starting state). \\[8pt]

            \multicolumn{2}{l}{\textbf{Transition Types}} \\
            \midrule
            \tikz\draw[->, red!70!black, thick] (0,0) -- (0.9,0); &
                \textbf{Inference executed:} $\mathcal{M}(f_t)$ is invoked to evaluate the current frame. \\[4pt]
            \tikz\draw[->, black, thick] (0,0) -- (0.9,0); &
                \textbf{Zero-inference step:} Frame is skipped, or state variables are updated seamlessly. \\[8pt]

            \multicolumn{2}{l}{\textbf{Variables \& Parameters}} \\
            \midrule
            $s_t \in \{0,1\}$
                & Skip module ($\mathcal{S}$) prediction ($1$: motion, $0$: static). \\[3pt]
            $\hat{y}_t \in \{0,1\}$
                & Classification model ($\mathcal{M}$) prediction ($1$: fire/smoke, $0$: safe). \\[3pt]
            $c_{\mathrm{fire}}$
                & Counter for consecutive positive detections ($\hat{y}_t=1$). \\[3pt]
            $c_{\mathrm{clear}}$
                & Counter for consecutive safe frames ($\hat{y}_t=0$) during EAGER mode. \\[3pt]
            $c_{\mathrm{skip}}$
                & Counter for consecutively skipped frames during NORMAL mode. \\[3pt]
            $c_{\mathrm{burst}}$
                & Countdown variable for remaining frames in a forced inference burst. \\[3pt]
            $W_{\mathrm{fire}}$
                & Threshold of consecutive positive detections required to trigger EAGER mode. \\[3pt]
            $W_{\mathrm{clr}}$
                & Threshold of consecutive negative frames required to return to NORMAL mode. \\[3pt]
            $N_{\mathrm{chk}}$
                & Maximum allowed skipped frames before triggering a forced burst. \\
        \end{tabular}
    };

\end{tikzpicture}
\end{document}
