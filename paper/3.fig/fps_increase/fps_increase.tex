\documentclass[tikz,border=6mm]{standalone}
\usepackage{xcolor}
\usepackage{pgfplots}
\usepackage{csvsimple-l3}
\pgfplotsset{compat=1.18}

\definecolor{clrBaseline}{RGB}{108,117,125}
\definecolor{clrSkip}{RGB}{13,110,253}
\definecolor{clrDL}{RGB}{220,53,69}
\definecolor{clrSkipMod}{RGB}{25,135,84}

% ─────────────────────────────────────────────
% Load CSV
% ─────────────────────────────────────────────
\csvreader[
    separator = semicolon,
]{fps_check_report.csv}{
    fps_no_skip        = \tmpFpsNoSkip,
    fps_skip           = \tmpFpsSkip,
    time_no_skip       = \tmpTimeNoSkip,
    time_skip          = \tmpTimeSkip,
    dl_avg_infer_ms    = \tmpTimeDL,
    skip_module_avg_ms = \tmpTimeSkipMod,
    run_ratio          = \tmpRunRatio,
    speedup            = \tmpSpeedup
}{%
    \global\let\fpsNoSkip\tmpFpsNoSkip
    \global\let\fpsSkip\tmpFpsSkip
    \global\let\timeNoSkip\tmpTimeNoSkip
    \global\let\timeSkip\tmpTimeSkip
    \global\let\timeDL\tmpTimeDL
    \global\let\timeSkipMod\tmpTimeSkipMod
    \global\let\runRatio\tmpRunRatio
    \global\let\speedup\tmpSpeedup
}

% Derived values
\pgfmathsetmacro{\timeDLinSkip}{\timeDL * \runRatio}
\pgfmathsetmacro{\speedupPct}{(\speedup - 1) * 100}
\pgfmathsetmacro{\ymaxFPS}{\fpsSkip * 1.3}
\pgfmathsetmacro{\ymaxTime}{\timeNoSkip * 1.35}

\pgfmathsetmacro{\timeRatio}{\timeDL / \timeSkipMod}
\pgfmathsetmacro{\timeRedPct}{100 * (\timeDL - \timeSkipMod) / \timeDL}

% Pre-formatted labels for stacked bar
\pgfmathprintnumberto[fixed,precision=1]{\timeDL}{\timeDLfmt}
\pgfmathprintnumberto[fixed,precision=1]{\timeDLinSkip}{\timeDLinSkipfmt}
\pgfmathprintnumberto[fixed,precision=2]{\timeSkipMod}{\timeSkipModfmt}


\begin{document}
\begin{tikzpicture}
    \tikzstyle{every node}=[font=\large]

    % ─────────────────────────────────────────
    % LEFT: Time Cost Decomposition
    % ─────────────────────────────────────────
    \begin{axis}[
        name             = ax1,
        ybar,
        bar shift        = 0pt,                      % ← center bars on their ticks
        bar width        = 1.8cm,
        width            = 7.5cm,
        height           = 7cm,
        symbolic x coords= {Big Model avg Inference, Skip method avg Proc},
        xtick            = {Big Model avg Inference, Skip method avg Proc},
        x tick label style = {
            font       = \small,
            align      = center,
            text width = 2.5cm,
        },
        ymin             = 0,
        ymax             = \ymaxTime,
        ylabel           = {Time per frame (ms)},
        title            = {(a) Time Budget},
        title style      = {font=\normalsize\bfseries, yshift=2pt},
        nodes near coords,
        every node near coord/.append style = {font=\small\bfseries},
        enlarge x limits = 0.4,
        label style      = {font=\normalsize},
        axis line style  = {gray!60},
        ymajorgrids      = true,
        grid style       = {dashed, gray!30},
        legend style     = {
            at           = {(0.5, -0.22)},
            anchor       = north,
            legend columns = 2,
            font         = \normalsize,
            draw         = none,
        },
    ]
        % Left bar: full DL inference time (no skip)
        \addplot[
            fill=clrDL, draw=clrDL!80!black,
            nodes near coords align=above,
            every node near coord/.append style={text=black}
        ]
            coordinates { (Big Model avg Inference, \timeDL) };

        % Right bar: skip module time ONLY
        \addplot[
            fill=clrSkipMod, draw=clrSkipMod!80!black,
            nodes near coords align=above,
            every node near coord/.append style={text=black}
        ]
            coordinates { (Skip method avg Proc, \timeSkipMod) };

        % 1. "x39 cheaper" horizontal link
        \draw[thick, clrSkipMod]
            (axis cs:Big Model avg Inference,\timeDL) |-
            ([yshift=0.8cm] axis cs:Big Model avg Inference,\timeDL) --
            node[above, font=\normalsize\bfseries, text=clrSkipMod] {$\times \pgfmathprintnumber[fixed,precision=0]{\timeRatio}$ cheaper}
            ([yshift=0.8cm] axis cs:Skip method avg Proc,\timeDL) -|
            ([yshift=0.5cm] axis cs:Skip method avg Proc,\timeSkipMod);

        % 2. Downward red arrow on the left bar
        \draw[stealth-, clrDL, ultra thick]
            ([{xshift=1.3cm}] axis cs:Big Model avg Inference,\timeSkipMod) --
            node[right, font=\normalsize\bfseries, text=clrDL] {$-$\pgfmathprintnumber[fixed,precision=1]{\timeRedPct}\% overhead}
            ([{xshift=1.3cm}] axis cs:Big Model avg Inference,\timeDL);

        % 3. Dashed horizontal line at right bar's height
        \draw[dashed, clrSkipMod, thick]
            ([{xshift=-0.9cm}] axis cs:Big Model avg Inference,\timeSkipMod) --
            ([{xshift=0.9cm}] axis cs:Skip method avg Proc,\timeSkipMod);
    \end{axis}

    % ─────────────────────────────────────────
    % RIGHT: Chart 1 — FPS Comparison
    % ─────────────────────────────────────────
    \begin{axis}[
        name             = ax2,
        at               = {(ax1.right of south east)},
        anchor           = left of south west,
        xshift           = 12mm,
        ybar,
        bar shift        = 0pt,
        bar width        = 1.8cm,
        width            = 7.5cm,
        height           = 7cm,
        symbolic x coords= {No-Skip, AccMotionDet},
        xtick            = {No-Skip, AccMotionDet},
        ymin             = 0,
        ymax             = \ymaxFPS,
        ylabel           = {Frames per Second (FPS)},
        title            = {(b) Throughput},
        title style      = {font=\normalsize\bfseries, yshift=2pt},
        nodes near coords,
        nodes near coords align   = {above},
        every node near coord/.append style = {font=\normalsize\bfseries},
        enlarge x limits = 0.4,
        tick label style = {font=\normalsize},
        label style      = {font=\normalsize},
        axis line style  = {gray!60},
        ymajorgrids      = true,
        grid style       = {dashed, gray!30},
    ]
        \addplot[fill=clrBaseline, draw=clrBaseline!80!black]
            coordinates {(No-Skip, \fpsNoSkip)};

        \addplot[fill=clrSkip, draw=clrSkip!80!black]
            coordinates {(AccMotionDet, \fpsSkip)};

        % Dashed baseline extension
        \draw[dashed, gray, thick]
            ([{xshift=-0.9cm}] axis cs:No-Skip,\fpsNoSkip) --
            ([{xshift=0.9cm}] axis cs:AccMotionDet,\fpsNoSkip);

        % Top boundary line for the arrow
        \draw[dashed, gray, thick]
            ([{xshift=-2.5cm}] axis cs:AccMotionDet,\fpsSkip) --
            ([{xshift=-0.9cm}] axis cs:AccMotionDet,\fpsSkip);

        % Speedup upward arrow
        \draw[latex-latex, clrSkipMod, very thick]
            ([{xshift=-1.4cm}] axis cs:AccMotionDet,\fpsNoSkip) --
            node[left, font=\normalsize\bfseries, text=clrSkip] {$+\pgfmathprintnumber[fixed,precision=1]{\speedupPct}\%$}
            ([{xshift=-1.4cm}] axis cs:AccMotionDet,\fpsSkip);
    \end{axis}


\end{tikzpicture}
